= 6. QUALIFY — 10 minutes

== Show

Submit the same three fixture classes through one deterministic contract.

== Learn

Complete evidence inside the envelope yields `ALLOW_REVIEW`; a known boundary
violation yields `REFUSE`; incomplete correlation yields `ABSTAIN`.
`HUMAN_PROMOTION_REQUIRED` remains the boundary and `llm_authority` remains
`NONE`.

== Do

[source,bash,role="execute"]
----
for case in qualified capacity gap; do
  curl -fsS -H 'Content-Type: application/json' --data-binary "@/tmp/fleet-${case}.json" \
    "$QUAL_URL/api/v1/qualifications" | tee "/tmp/fleet-${case}-result.json" | \
    jq '{request_id,decision,source_state,reason_codes,authority,qualification}'
done
----

== Prove

[source,bash,role="execute"]
----
test "$(jq -r .decision /tmp/fleet-qualified-result.json)" = ALLOW_REVIEW
test "$(jq -r .decision /tmp/fleet-capacity-result.json)" = REFUSE
test "$(jq -r .decision /tmp/fleet-gap-result.json)" = ABSTAIN
test "$(jq -r .authority.llm_authority /tmp/fleet-qualified-result.json)" = NONE
----
